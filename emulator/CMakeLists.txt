cmake_minimum_required(VERSION 3.16)
project(emulator CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native")
find_package(OpenMP)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_emu src/bindings.cpp src/solver.cpp)
if(OpenMP_CXX_FOUND)
  target_link_libraries(_emu PRIVATE OpenMP::OpenMP_CXX)
endif()
set_target_properties(_emu PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})
